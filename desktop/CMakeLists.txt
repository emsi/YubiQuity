cmake_minimum_required(VERSION 3.22)
project(yubiquity-desktop VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
add_library(desktop-ui STATIC ui.cpp process.cpp window.cpp presentation.cpp setup.cpp)
target_link_libraries(desktop-ui PUBLIC Qt6::Widgets Qt6::DBus)
target_link_libraries(desktop-ui PRIVATE PkgConfig::XCB)
target_compile_options(desktop-ui PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(desktop-ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(yubiquity-desktop main.cpp)
target_link_libraries(yubiquity-desktop PRIVATE desktop-ui)
target_compile_definitions(yubiquity-desktop PRIVATE YUBIQUITY_VERSION="${PROJECT_VERSION}")
target_compile_options(yubiquity-desktop PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Network Test)
  add_executable(desktop-tests tests.cpp)
  target_link_libraries(desktop-tests PRIVATE desktop-ui Qt6::Test Qt6::Network)
  target_compile_options(desktop-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME desktop-tests COMMAND desktop-tests)
  set_tests_properties(desktop-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS yubiquity-desktop RUNTIME DESTINATION bin)
install(FILES org.yubiquity.Desktop.desktop DESTINATION share/applications)
